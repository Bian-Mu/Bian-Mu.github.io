#### 快慢指针的判圈

1. 快指针每次走两步，当与慢指针相遇则证明进入了环，若到达终点则无环（相遇点到环入口的距离与起点到环入口的距离相差整数倍环的长度）
2. 此时将慢指针拨回起点，两个指针以同样的速度移动，再次相遇时为环的入口

```js
function buildMaxHeap(nums,maxSize){
	for(let i=Math.floor(maxSize/2-1);i>=0;i--){
		maxHeapify(i,nums,maxSize)
	}

//取出元素后进行调整，使最大元素位于root
function maxHeapifyDown(index,nums,maxSize){
	while(index*2+1<maxSize){
		let left=2*index+1,right=2*index+2
		let Largest=index

		if(left<maxSize&&nums[left]>nums[Largest]){
			Largest=left
		}

		if(right<maxSize&&nums[right]>nums[Largest]){
			Largest=right
		}

		if(Largest!==index){
			[nums[index],nums[Largest]]=[nums[Largest],nums[index]]
			index=Largest
		}else{
			break
		}
	}
}

//插入元素后调整
function maxHeapifyUp(index,nums){
	while(index>0){
		let parent=Math.floor(index/2-1)
		if(nums[parent]<nums[index]){
			[nums[index],nums[parent]]=[nums[parent],nums[index]]
			index=parent
		}else{
			break
		}
	}
}
```

```ts
function getMinimumDifference(root: TreeNode | null): number {
	const stack: TreeNode[] = []
	let prev = -1
	let ans = Infinity
	while (root || stack.length) {
		while (root) {
			stack.push(root)
			root = root.left
		}
		root = stack.pop()!
		if (prev === -1) {
			prev = root.val
		} else {
			ans = Math.min(ans, root.val - prev)
			prev = root.val
		}
		root = root.right
	}
	return ans
};
```

### 回溯 backtrack

多半是dfs，重点在于递归时要动态保持当前的状态

```ts
function permute(nums: number[]): number[][] {

	const ans = new Array<number[]>()

	const backtrack = (curr: number, temp: number[]) => {
		if (curr === nums.length) {
			ans.push([...temp])
			return
		}

		for (let i = curr; i < nums.length; i++) {
			[temp[i], temp[curr]] = [temp[curr], temp[i]]
			backtrack(curr + 1, temp);
			[temp[i], temp[curr]] = [temp[curr], temp[i]]
		}
	}

	backtrack(0, nums)

	return ans
};
```
